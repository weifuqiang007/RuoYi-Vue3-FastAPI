### 1.1 补充知识库更新和删除接口

在 `module_rag/controller/knowledge_base_controller.py` 中添加两个端点。

> **说明**：Service 层的 `update()` 和 `delete()` 方法已经存在于 `module_rag/service/knowledge_base_service.py`，DAO 层也已实现，只需添加 Controller 端点即可。

**完整文件内容**（直接替换 `knowledge_base_controller.py`）：

```python
# module_rag/controller/knowledge_base_controller.py
from typing import Annotated
from fastapi import Body, Path
from sqlalchemy.ext.asyncio import AsyncSession
from common.aspect.db_seesion import DBSessionDependency
from common.aspect.pre_auth import CurrentUserDependency, PreAuthDependency
from common.router import APIRouterPro
from exceptions.exception import ServiceException
from module_admin.entity.vo.user_vo import CurrentUserModel
from module_rag.entity.vo.knowledge_base_vo import KnowledgeBaseCreateModel, KnowledgeBaseUpdateModel
from module_rag.service.knowledge_base_service import KnowledgeBaseService
from utils.response_util import ResponseUtil

knowledge_base_controller = APIRouterPro(
    prefix='/rag/kb', order_num=20, tags=['RAG管理-知识库'], dependencies=[PreAuthDependency()]
)

# 允许访问的角色
ALLOWED_ROLES = {'admin', 'teacher'}


def check_role(current_user: CurrentUserModel):
    """检查当前用户是否有权限访问 RAG 管理接口"""
    roles = set(current_user.roles) if current_user.roles else set()
    if not roles.intersection(ALLOWED_ROLES):
        raise ServiceException(message='无权访问，仅管理员和教师可操作知识库')


@knowledge_base_controller.post('', summary='创建知识库')
async def create_kb(
    data: KnowledgeBaseCreateModel,
    query_db: Annotated[AsyncSession, DBSessionDependency()],
    current_user: CurrentUserModel = CurrentUserDependency(),
):
    check_role(current_user)
    kb = await KnowledgeBaseService.create(query_db, data, current_user.user.user_id)
    return ResponseUtil.success(data={"kb_id": kb.kb_id})


@knowledge_base_controller.get('/list', summary='获取知识库列表')
async def get_kb_list(
    query_db: Annotated[AsyncSession, DBSessionDependency()],
    current_user: CurrentUserModel = CurrentUserDependency(),
):
    check_role(current_user)
    kb_list = await KnowledgeBaseService.get_list(query_db)
    return ResponseUtil.success(data=[{
        "kb_id": kb.kb_id,
        "kb_name": kb.kb_name,
        "kb_desc": kb.kb_desc,
        "embedding_model": kb.embedding_model,
        "chunk_size": kb.chunk_size,
        "chunk_overlap": kb.chunk_overlap,
        "doc_count": kb.doc_count,
        "create_time": str(kb.create_time),
    } for kb in kb_list])


@knowledge_base_controller.get('/{kb_id}', summary='获取知识库详情')
async def get_kb_detail(
    kb_id: int,
    query_db: Annotated[AsyncSession, DBSessionDependency()],
    current_user: CurrentUserModel = CurrentUserDependency(),
):
    check_role(current_user)
    kb = await KnowledgeBaseService.get_by_id(query_db, kb_id)
    if not kb:
        return ResponseUtil.failure(msg="知识库不存在")
    return ResponseUtil.success(data={
        "kb_id": kb.kb_id,
        "kb_name": kb.kb_name,
        "kb_desc": kb.kb_desc,
        "embedding_model": kb.embedding_model,
        "chunk_size": kb.chunk_size,
        "chunk_overlap": kb.chunk_overlap,
        "doc_count": kb.doc_count,
        "status": kb.status,
        "create_time": str(kb.create_time),
    })


@knowledge_base_controller.put('', summary='更新知识库')
async def update_kb(
    data: KnowledgeBaseUpdateModel,
    query_db: Annotated[AsyncSession, DBSessionDependency()],
    current_user: CurrentUserModel = CurrentUserDependency(),
):
    check_role(current_user)
    await KnowledgeBaseService.update(query_db, data)
    return ResponseUtil.success(msg="更新成功")


@knowledge_base_controller.delete('/{kb_ids}', summary='删除知识库')
async def delete_kb(
    kb_ids: str,
    query_db: Annotated[AsyncSession, DBSessionDependency()],
    current_user: CurrentUserModel = CurrentUserDependency(),
):
    check_role(current_user)
    ids = kb_ids.split(',')
    for kb_id in ids:
        await KnowledgeBaseService.delete(query_db, int(kb_id))
    return ResponseUtil.success(msg="删除成功")
```

```vue
        </el-form-item>

        </el-form-item>
      </el-form>
      <template #footer>
        <el-button @click="dialogVisible = false">取 消</el-button>
        <el-button type="primary" @click="submitForm">确 定</el-button>
      </template>
    </el-dialog>
  </div>

<script setup>
import { ref, reactive, onMounted } from 'vue'
import { useRouter } from 'vue-router'
import { ElMessage, ElMessageBox } from 'element-plus'
import {
  listKnowledgeBase,
  addKnowledgeBase,
  updateKnowledgeBase,
  delKnowledgeBase
} from '@/api/rag/knowledgeBase'

const router = useRouter()
const loading = ref(false)
const kbList = ref([])
const dialogVisible = ref(false)
const dialogTitle = ref('')
const formRef = ref(null)

const form = reactive({
  kb_id: undefined,
  kb_name: '',
  kb_desc: '',
  chunk_size: 500,
  chunk_overlap: 50
})

const rules = {
  kb_name: [{ required: true, message: '请输入知识库名称', trigger: 'blur' }]
}

/** 获取列表 */
function getList() {
  loading.value = true
  listKnowledgeBase().then(res => {
    kbList.value = res.data
    loading.value = false
  })
}

/** 新增 */
function handleAdd() {
  resetForm()
  dialogTitle.value = '新增知识库'
  dialogVisible.value = true
}

/** 编辑 */
function handleEdit(row) {
  resetForm()
  Object.assign(form, row)
  dialogTitle.value = '修改知识库'
  dialogVisible.value = true
}

/** 提交 */
function submitForm() {
  formRef.value.validate(valid => {
    if (!valid) return
    if (form.kb_id) {
      updateKnowledgeBase(form).then(() => {
        ElMessage.success('修改成功')
        dialogVisible.value = false
        getList()
      })
    } else {
      addKnowledgeBase(form).then(() => {
        ElMessage.success('新增成功')
        dialogVisible.value = false
        getList()
      })
    }
  })
}

/** 删除 */
function handleDelete(row) {
  ElMessageBox.confirm(`确认删除知识库「${row.kb_name}」？`, '提示', {
    confirmButtonText: '确定',
    cancelButtonText: '取消',
    type: 'warning'
  }).then(() => {
    delKnowledgeBase(row.kb_id).then(() => {
      ElMessage.success('删除成功')
      getList()
    })
  }).catch(() => {})
}

/** 进入文档管理 */
function handleDocument(row) {
  router.push({ path: `/rag/document/${row.kb_id}` })
}

/** 行点击也进入文档管理 */
function handleRowClick(row) {
  handleDocument(row)
}

function resetForm() {
  form.kb_id = undefined
  form.kb_name = ''
  form.kb_desc = ''
  form.chunk_size = 500
  form.chunk_overlap = 50
}

onMounted(() => {
  getList()
})
</script>
```

```vue
<script setup>
import { ref, computed, onMounted } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import { ElMessage, ElMessageBox } from 'element-plus'
import { listDocument, delDocument } from '@/api/rag/document'
import { getToken } from '@/utils/auth'

const route = useRoute()
const router = useRouter()
const kbId = computed(() => route.params.kbId)
const loading = ref(false)
const docList = ref([])

const uploadUrl = computed(() => `${import.meta.env.VITE_APP_BASE_API}/rag/document/upload/${kbId.value}`)
const uploadHeaders = computed(() => ({ Authorization: `Bearer ${getToken()}` }))

/** 获取文档列表 */
function getList() {
  loading.value = true
  listDocument(kbId.value).then(res => {
    docList.value = res.data
    loading.value = false
  })
}

/** 上传前校验 */
function beforeUpload(file) {
  const allowed = ['.pdf', '.docx', '.doc', '.txt', '.md']
  const ext = file.name.substring(file.name.lastIndexOf('.')).toLowerCase()
  if (!allowed.includes(ext)) {
    ElMessage.error('仅支持 PDF、Word、TXT、Markdown 格式')
    return false
  }
  return true
}

/** 上传成功 */
function handleUploadSuccess() {
  ElMessage.success('上传成功，正在处理中...')
  // 延迟刷新，等待后端处理
  setTimeout(() => getList(), 2000)
}

/** 上传失败 */
function handleUploadError() {
  ElMessage.error('上传失败')
}

/** 删除 */
function handleDelete(row) {
  ElMessageBox.confirm(`确认删除文档「${row.doc_name}」？`, '提示', {
    confirmButtonText: '确定',
    cancelButtonText: '取消',
    type: 'warning'
  }).then(() => {
    delDocument(row.doc_id).then(() => {
      ElMessage.success('删除成功')
      getList()
    })
  }).catch(() => {})
}

/** 返回 */
function goBack() {
  router.push('/rag/kb')
}

/** 状态映射 */
function parseStatusLabel(s) {
  return { '0': '待处理', '1': '解析中', '2': '完成', '9': '失败' }[s] || '未知'
}
function parseStatusType(s) {
  return { '0': 'info', '1': 'warning', '2': 'success', '9': 'danger' }[s] || 'info'
}
function embedStatusLabel(s) {
  return { '0': '待处理', '1': '向量化中', '2': '完成', '9': '失败' }[s] || '未知'
}
function embedStatusType(s) {
  return { '0': 'info', '1': 'warning', '2': 'success', '9': 'danger' }[s] || 'info'
}

/** 文件大小格式化 */
function formatFileSize(bytes) {
  if (!bytes) return '0 B'
  const units = ['B', 'KB', 'MB', 'GB']
  let i = 0
  let size = bytes
  while (size >= 1024 && i < units.length - 1) {
    size /= 1024
    i++
  }
  return size.toFixed(1) + ' ' + units[i]
}

onMounted(() => {
  getList()
})
```

### 3.1 知识库管理接口

| # | Method | Path | 说明 | 权限 |
|---|---|---|---|---|
| 1 | POST | `/rag/kb` | 创建知识库 | admin/教师 |
| 2 | GET | `/rag/kb/list` | 知识库列表 | admin/教师 |
| 3 | GET | `/rag/kb/{kb_id}` | 知识库详情 | admin/教师 |
| 4 | PUT | `/rag/kb` | 更新知识库 | admin/教师 |
| 5 | DELETE | `/rag/kb/{kb_ids}` | 删除知识库（逗号分隔多ID） | admin/教师 |

**POST `/rag/kb` — 创建知识库**

Request Body:
```json
{
  "kb_name": "社会工作理论知识库",    // 必填
  "kb_desc": "描述信息",             // 可选
  "embedding_model": "embedding-3",  // 可选，默认 embedding-3
  "chunk_size": 500,                 // 可选，默认 500
  "chunk_overlap": 50                // 可选，默认 50
}
```

Response:
```json
{
  "code": 200,
  "msg": "操作成功",
  "data": { "kb_id": 5 }
}
```

**GET `/rag/kb/list` — 知识库列表**

Response:
```json
{
  "code": 200,
  "data": [
    {
      "kb_id": 1,
      "kb_name": "社会工作理论知识库",
      "kb_desc": "包含社会工作教材、论文、伦理守则等",
      "embedding_model": "embedding-3",
      "chunk_size": 500,
      "chunk_overlap": 50,
      "doc_count": 3,
      "create_time": "2026-06-04 10:00:00"
    }
  ]
}
```

**PUT `/rag/kb` — 更新知识库**

Request Body:
```json
{
  "kb_id": 1,               // 必填
  "kb_name": "新名称",       // 可选
  "kb_desc": "新描述",       // 可选
  "chunk_size": 600,         // 可选
  "chunk_overlap": 80        // 可选
}
```

Response:
```json
{ "code": 200, "msg": "更新成功" }
```

**DELETE `/rag/kb/{kb_ids}` — 删除知识库**

示例: `DELETE /rag/kb/1,2,3`

Response:
```json
{ "code": 200, "msg": "删除成功" }
```

### 3.2 文档管理接口

| # | Method | Path | 说明 | 权限 |
|---|---|---|---|---|
| 1 | POST | `/rag/document/upload/{kb_id}` | 上传文档（multipart/form-data） | admin/教师 |
| 2 | GET | `/rag/document/list/{kb_id}` | 文档列表 | admin/教师 |
| 3 | GET | `/rag/document/download/{doc_id}` | 下载文档 | admin/教师 |
| 4 | GET | `/rag/document/preview/{doc_id}` | 获取预览URL | admin/教师 |
| 5 | DELETE | `/rag/document/{doc_ids}` | 删除文档（逗号分隔多ID） | admin/教师 |

**POST `/rag/document/upload/{kb_id}` — 上传文档**

Request: `multipart/form-data`，字段名 `file`，支持 `.pdf/.docx/.doc/.txt/.md`

Response:
```json
{
  "code": 200,
  "data": { "doc_id": 10, "file_path": "rag/1/abc123.pdf" }
}
```

**GET `/rag/document/list/{kb_id}` — 文档列表**

Response:
```json
{
  "code": 200,
  "data": [
    {
      "doc_id": 10,
      "doc_name": "社会工作伦理守则.pdf",
      "file_type": ".pdf",
      "file_size": 245678,
      "parse_status": "2",
      "embed_status": "2",
      "chunk_count": 15
    }
  ]
}
```

状态码说明：
- `parse_status`: `0`=待处理, `1`=解析中, `2`=完成, `9`=失败
- `embed_status`: `0`=待处理, `1`=向量化中, `2`=完成, `9`=失败
