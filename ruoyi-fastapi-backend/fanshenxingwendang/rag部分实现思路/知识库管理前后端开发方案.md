## 一、后端工程师任务

> **核心设计原则**：
>
> 1. **角色校验采用 AOP 路由级依赖**：利用项目已有的 `RoleInterfaceAuthDependency`，声明在 `APIRouterPro` 的 `dependencies` 中，所有注册在该路由下的接口自动生效，无需在每个端点函数内手动调用 `check_role`。
> 2. **控制器采用类形式（CBV）**：用 `@staticmethod` + 路由装饰器将增删改查接口组织在同一个类中，提升可维护性。路由实例仍保持模块级变量，确保 `RouterRegister` 自动注册机制正常工作。
> 3. **响应序列化使用 Pydantic Model**：用 `KnowledgeBaseResponseModel.model_validate()` + `@field_serializer` 替代手动拼字典，类型安全且易维护。

### 1.1 知识库 Controller（类形式 + AOP 角色校验）

在 `module_rag/controller/knowledge_base_controller.py` 中，补齐更新、详情、删除端点，并将所有接口以类形式组织。

> **说明**：Service 层的 `update()` 和 `delete()` 方法已经存在于 `module_rag/service/knowledge_base_service.py`，DAO 层也已实现，只需添加 Controller 端点即可。

**完整文件内容**（直接替换 `knowledge_base_controller.py`）：

```python
# module_rag/controller/knowledge_base_controller.py
from typing import Annotated

from fastapi import Path
from sqlalchemy.ext.asyncio import AsyncSession

from common.aspect.db_seesion import DBSessionDependency
from common.aspect.interface_auth import RoleInterfaceAuthDependency
from common.aspect.pre_auth import CurrentUserDependency, PreAuthDependency
from common.router import APIRouterPro
from module_admin.entity.vo.user_vo import CurrentUserModel
from module_rag.entity.vo.knowledge_base_vo import (
    KnowledgeBaseCreateModel,
    KnowledgeBaseResponseModel,
    KnowledgeBaseUpdateModel,
)
from module_rag.service.knowledge_base_service import KnowledgeBaseService
from utils.response_util import ResponseUtil

# ──────────────────────────────────────────────
# 路由定义
# 角色校验通过路由级依赖实现 AOP 效果，
# 所有注册在该路由下的接口自动生效，无需手动调用 check_role
# ──────────────────────────────────────────────
knowledge_base_controller = APIRouterPro(
    prefix='/rag/kb',
    order_num=20,
    tags=['RAG管理-知识库'],
    dependencies=[
        PreAuthDependency(),
        RoleInterfaceAuthDependency(['admin', 'teacher']),
    ],
)


class KnowledgeBaseController:
    """知识库管理控制器 —— 将增删改查接口以类形式组织"""

    # ── 创建 ──────────────────────────────────
    @staticmethod
    @knowledge_base_controller.post('', summary='创建知识库')
    async def create(
        data: KnowledgeBaseCreateModel,
        query_db: Annotated[AsyncSession, DBSessionDependency()],
        current_user: CurrentUserModel = CurrentUserDependency(),
    ):
        kb = await KnowledgeBaseService.create(query_db, data, current_user.user.user_id)
        return ResponseUtil.success(data={'kb_id': kb.kb_id})

    # ── 列表 ──────────────────────────────────
    @staticmethod
    @knowledge_base_controller.get('/list', summary='获取知识库列表')
    async def get_list(
        query_db: Annotated[AsyncSession, DBSessionDependency()],
    ):
        kb_list = await KnowledgeBaseService.get_list(query_db)
        return ResponseUtil.success(
            data=[KnowledgeBaseResponseModel.model_validate(kb).model_dump() for kb in kb_list]
        )

    # ── 详情 ──────────────────────────────────
    @staticmethod
    @knowledge_base_controller.get('/{kb_id}', summary='获取知识库详情')
    async def get_detail(
        kb_id: Annotated[int, Path(description='知识库ID')],
        query_db: Annotated[AsyncSession, DBSessionDependency()],
    ):
        kb = await KnowledgeBaseService.get_by_id(query_db, kb_id)
        if not kb:
            return ResponseUtil.failure(msg='知识库不存在')
        return ResponseUtil.success(
            data=KnowledgeBaseResponseModel.model_validate(kb).model_dump()
        )

    # ── 更新 ──────────────────────────────────
    @staticmethod
    @knowledge_base_controller.put('', summary='更新知识库')
    async def update(
        data: KnowledgeBaseUpdateModel,
        query_db: Annotated[AsyncSession, DBSessionDependency()],
    ):
        await KnowledgeBaseService.update(query_db, data)
        return ResponseUtil.success(msg='更新成功')

    # ── 删除 ──────────────────────────────────
    @staticmethod
    @knowledge_base_controller.delete('/{kb_ids}', summary='删除知识库')
    async def delete(
        kb_ids: Annotated[str, Path(description='知识库ID，多个用逗号分隔')],
        query_db: Annotated[AsyncSession, DBSessionDependency()],
    ):
        for kb_id in kb_ids.split(','):
            await KnowledgeBaseService.delete(query_db, int(kb_id))
        return ResponseUtil.success(msg='删除成功')
```

**完整文件内容**（直接替换 `document_controller.py`）：

```python
# module_rag/controller/document_controller.py
import os
import uuid
from typing import Annotated, Optional

from fastapi import File, Query, UploadFile
from fastapi.responses import StreamingResponse
from sqlalchemy.ext.asyncio import AsyncSession

from common.aspect.db_seesion import DBSessionDependency
from common.aspect.interface_auth import RoleInterfaceAuthDependency
from common.aspect.pre_auth import CurrentUserDependency, PreAuthDependency
from common.router import APIRouterPro
from module_admin.entity.vo.user_vo import CurrentUserModel
from module_rag.dao.document_dao import DocumentDao
from module_rag.entity.do.document_do import RagDocument
from module_rag.service.document_service import DocumentService
from module_rag.utils.minio_client import MinioClient
from utils.response_util import ResponseUtil

document_controller = APIRouterPro(
    prefix='/rag/document',
    order_num=21,
    tags=['RAG管理-文档'],
    dependencies=[
        PreAuthDependency(),
        RoleInterfaceAuthDependency(['admin', 'teacher']),
    ],
)


class DocumentController:
    """文档管理控制器 —— 支持独立页面模式（全部文档 + 按知识库筛选）"""

    # ── 全部文档列表（新增，支持可选 kb_id 筛选）──────────
    @staticmethod
    @document_controller.get('/list', summary='获取全部文档列表')
    async def get_all_doc_list(
        query_db: Annotated[AsyncSession, DBSessionDependency()],
        kb_id: Optional[int] = Query(None, description='按知识库ID筛选，不传则返回全部'),
    ):
        """
        获取全部文档列表，支持按知识库筛选。
        - 不传 kb_id：返回所有文档
        - 传 kb_id：返回指定知识库下的文档
        """
        if kb_id is not None:
            docs = await DocumentDao.get_by_kb_id(query_db, kb_id)
        else:
            docs = await DocumentDao.get_all(query_db)
        return ResponseUtil.success(data=[{
            "doc_id": d.doc_id,
            "kb_id": d.kb_id,
            "doc_name": d.doc_name,
            "file_type": d.file_type,
            "file_size": d.file_size,
            "parse_status": d.parse_status,
            "embed_status": d.embed_status,
            "chunk_count": d.chunk_count,
            "create_time": str(d.create_time) if d.create_time else None,
        } for d in docs])

    # ── 某个知识库下的文档列表（保留兼容）──────────
    @staticmethod
    @document_controller.get('/list/{kb_id}', summary='获取知识库下的文档列表')
    async def get_doc_list(
        kb_id: int,
        query_db: Annotated[AsyncSession, DBSessionDependency()],
    ):
        docs = await DocumentDao.get_by_kb_id(query_db, kb_id)
        return ResponseUtil.success(data=[{
            "doc_id": d.doc_id,
            "kb_id": d.kb_id,
            "doc_name": d.doc_name,
            "file_type": d.file_type,
            "file_size": d.file_size,
            "parse_status": d.parse_status,
            "embed_status": d.embed_status,
            "chunk_count": d.chunk_count,
            "create_time": str(d.create_time) if d.create_time else None,
        } for d in docs])

    # ── 上传文档到知识库 ──────────────────────────
    @staticmethod
    @document_controller.post('/upload/{kb_id}', summary='上传文档到知识库')
    async def upload_document(
        kb_id: int,
        file: UploadFile = File(...),
        query_db: Annotated[AsyncSession, DBSessionDependency()] = None,
        current_user: CurrentUserModel = CurrentUserDependency(),
    ):
        content = await file.read()
        file_type = os.path.splitext(file.filename)[1].lower()

        # 上传到 MinIO
        object_name = f"rag/{kb_id}/{uuid.uuid4().hex}{file_type}"
        minio = MinioClient.get_instance()
        minio.put(object_name, content)

        # 创建文档记录
        doc = RagDocument(
            kb_id=kb_id,
            doc_name=file.filename,
            file_path=object_name,
            file_type=file_type,
            file_size=len(content),
            user_id=current_user.user.user_id,
        )
        doc = await DocumentDao.create(query_db, doc)

        # 触发异步处理
        await DocumentService.process_document(query_db, doc.doc_id)

        return ResponseUtil.success(data={"doc_id": doc.doc_id, "file_path": object_name})

    # ── 下载文档 ──────────────────────────────────
    @staticmethod
    @document_controller.get('/download/{doc_id}', summary='下载文档')
    async def download_document(
        doc_id: int,
        query_db: Annotated[AsyncSession, DBSessionDependency()],
    ):
        doc = await DocumentDao.get_by_id(query_db, doc_id)
        if not doc:
            return ResponseUtil.failure(msg="文档不存在")

        minio = MinioClient.get_instance()
        data = minio.get(doc.file_path)
        if data is None:
            return ResponseUtil.failure(msg="文件不存在或已损坏")

        import io
        return StreamingResponse(
            io.BytesIO(data),
            media_type="application/octet-stream",
            headers={"Content-Disposition": f"attachment; filename*=UTF-8''{doc.doc_name}"},
        )

    # ── 文档预览 ──────────────────────────────────
    @staticmethod
    @document_controller.get('/preview/{doc_id}', summary='获取文档预览URL')
    async def preview_document(
        doc_id: int,
        query_db: Annotated[AsyncSession, DBSessionDependency()],
    ):
        doc = await DocumentDao.get_by_id(query_db, doc_id)
        if not doc:
            return ResponseUtil.failure(msg="文档不存在")

        minio = MinioClient.get_instance()
        url = minio.get_presigned_url(doc.file_path)
        if url is None:
            return ResponseUtil.failure(msg="获取预览地址失败")

        return ResponseUtil.success(data={"url": url, "doc_name": doc.doc_name})

    # ── 删除文档 ──────────────────────────────────
    @staticmethod
    @document_controller.delete('/{doc_ids}', summary='删除文档')
    async def delete_document(
        doc_ids: str,
        query_db: Annotated[AsyncSession, DBSessionDependency()],
    ):
        ids = [int(i) for i in doc_ids.split(',')]
        minio = MinioClient.get_instance()

        for doc_id in ids:
            doc = await DocumentDao.get_by_id(query_db, doc_id)
            if doc:
                minio.remove(doc.file_path)
                await DocumentDao.update_status(query_db, doc_id, del_flag='2')

        return ResponseUtil.success(msg="删除成功")
```

# 在 DocumentDao 类中新增：
@classmethod
async def get_all(cls, db: AsyncSession) -> list[RagDocument]:
    """获取全部未删除的文档"""
    result = await db.execute(
        select(RagDocument)
        .where(RagDocument.del_flag == '0')
        .order_by(RagDocument.create_time.desc())
    )
    return list(result.scalars().all())

import { ref, onMounted } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import { listAllDocument, uploadDocument, downloadDocument, delDocument } from '@/api/rag/document'
import { listKnowledgeBase } from '@/api/rag/knowledgeBase'

const docList = ref([])
const kbOptions = ref([])
const selectedKbId = ref(null)

const uploadKbId = ref(null)

const uploading = ref(false)
const uploadRef = ref(null)

    kbOptions.value = res.data

/** 获取文档列表 */

}

/** 根据kb_id获取知识库名称 */
function getKbName(kbId) {
  const kb = kbOptions.value.find(k => k.kb_id === kbId)
  return kb ? kb.kb_name : `ID: ${kbId}`
}

/** 点击上传按钮 */

  uploadKbId.value = selectedKbId.value || null
  uploadFile.value = null
  uploadDialogVisible.value = true
}

/** 文件选择变化 */
function handleFileChange(file) {
  const allowed = ['.pdf', '.docx', '.doc', '.txt', '.md']
  const ext = file.name.substring(file.name.lastIndexOf('.')).toLowerCase()
  if (!allowed.includes(ext)) {
    ElMessage.error('仅支持 PDF、Word、TXT、Markdown 格式')
    uploadRef.value?.clearFiles()
    return
  }
  uploadFile.value = file.raw
}

/** 文件移除 */
function handleFileRemove() {
  uploadFile.value = null
}

/** 提交上传 */
async function submitUpload() {
  if (!uploadKbId.value) {
    ElMessage.warning('请选择目标知识库')
    return
  }
  if (!uploadFile.value) {
    ElMessage.warning('请选择文件')
    return
  }
  uploading.value = true
  try {
    await uploadDocument(uploadKbId.value, uploadFile.value)
    ElMessage.success('上传成功，正在处理中...')
    uploadDialogVisible.value = false
    // 延迟刷新，等待后端处理
    setTimeout(() => getList(), 2000)
  } catch {
    ElMessage.error('上传失败')
  } finally {
    uploading.value = false
  }
}

async function handleDownload(row) {
  try {
    const res = await downloadDocument(row.doc_id)
    const blob = new Blob([res])
    const url = window.URL.createObjectURL(blob)
    const link = document.createElement('a')
    link.href = url
    link.download = row.doc_name
    link.click()
    window.URL.revokeObjectURL(url)
  } catch {
    ElMessage.error('下载失败')
  }
}

/** 删除文档 */
function handleDelete(row) {
  ElMessageBox.confirm(`确认删除文档「${row.doc_name}」？`, '提示', {
    confirmButtonText: '确定',
    cancelButtonText: '取消',
    type: 'warning'

    delDocument(row.doc_id).then(() => {
      ElMessage.success('删除成功')
      getList()

}

function parseStatusLabel(s) {
  return { '0': '待处理', '1': '解析中', '2': '完成', '9': '失败' }[s] || '未知'
}
function parseStatusType(s) {
  return { '0': 'info', '1': 'warning', '2': 'success', '9': 'danger' }[s] || 'info'
}
function embedStatusLabel(s) {
  return { '0': '待处理', '1': '向量化中', '2': '完成', '9': '失败' }[s] || '未知'
}
function embedStatusType(s) {
  return { '0': 'info', '1': 'warning', '2': 'success', '9': 'danger' }[s] || 'info'
}

/** 文件大小格式化 */
function formatFileSize(bytes) {
  if (!bytes) return '0 B'
  const units = ['B', 'KB', 'MB', 'GB']
  let i = 0
  let size = bytes
  while (size >= 1024 && i < units.length - 1) {
    size /= 1024
    i++
  }
  return size.toFixed(1) + ' ' + units[i]
}

onMounted(() => {
